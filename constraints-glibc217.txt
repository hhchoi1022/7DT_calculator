# Version ceiling for 7ds (CentOS 7, glibc 2.17, GCC 4.8.5).
#
# Newer releases of these packages ship only manylinux_2_28 wheels, so pip falls
# back to building from source and NumPy then refuses the 4.8.5 compiler. Every
# pin below is the newest release that still publishes a manylinux2014 wheel,
# and all of them satisfy requirements.txt.
#
# Install with --only-binary limited to these names, so a missing wheel among
# them fails loudly instead of starting a compile that cannot succeed. It is not
# :all:, because astroplan publishes no wheel above 0.6 — it is pure Python and
# builds from its sdist without a compiler.
numpy==1.26.4
scipy==1.12.0
pandas==2.2.3
matplotlib==3.8.4
shapely==2.0.6
astropy==7.1.0
pyarrow==15.0.2
